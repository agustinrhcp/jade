01

Values and primitives

Int, Float, String, Char and Bool. Annotations go on the signature; inside a function the compiler works the rest out. A zero-argument function is a value where you reference it: total, not total(), though the parentheses stay at the definition so it is greppable.

values.jd 
module Values exposing (greeting, initial, pi, ready, total)

def total() -> Int
  6 * 7
end


def pi() -> Float
  3.14159
end


def greeting() -> String
  "hello" ++ ", " ++ "world"
end


def initial() -> Char
  'J'
end


def ready() -> Bool
  total > 0
end
02

Functions

Arguments and returns are annotated; the body is one expression, so there is no return. ++ joins strings and arrays alike.

functions.jd 
module Functions exposing (full_name, shout)

def full_name(first: String, last: String) -> String
  first ++ " " ++ last
end


def shout(first: String, last: String) -> String
  full_name(first, last)
    |> String.to_upper
end
03

Union types

One type, a fixed set of shapes, each carrying its own data. Where Ruby would reach for a status string or a subclass per case, this is checked. Add a shape and the compiler walks you round every place that has to handle it.

unions.jd 
module Unions exposing (describe, radius_of)

type Shape
  = Circle(Float)
  | Rectangle(Float, Float)
  | Point


def describe(shape: Shape) -> String
  case shape
  in Circle(_) then "circle"
  in Rectangle(_, _) then "rectangle"
  in Point then "point"
  end
end


def radius_of(shape: Shape) -> Maybe(Float)
  case shape
  in Circle(r) then Just(r)
  in Rectangle(_, _) then Nothing
  in Point then Nothing
  end
end
04

Structs and records

A typed Struct, near enough. { p | age: … } is #with, a copy with one field replaced. Nothing mutates, so nothing changes underneath a caller holding the old one.

structs.jd 
module Structs exposing (birthday, label, rename)

struct Person = {
  name: String,
  age: Int
}


def birthday(p: Person) -> Person
  { p | age: p.age + 1 }
end


def rename(p: Person, name: String) -> Person
  { p | name: name }
end


def label(p: Person) -> String
  p.name ++ " (" ++ String.from_int(p.age) ++ ")"
end
05

Pattern matching

Ruby's case/in, with the compiler counting. It pulls values out as it branches and refuses to build if you have missed a shape. No else that quietly swallows the case you forgot.

matching.jd 
module Matching exposing (first_word, sum, total_of)

type Payment
  = Cash(Int)
  | Card(last4: String, cents: Int)


def total_of(p: Payment) -> Int
  case p
  in Cash(c) then c
  in Card(d) then d.cents
  end
end


def sum(xs: List(Int)) -> Int
  case xs
  in [] then 0
  in [x | rest] then x + sum(rest)
  end
end


def first_word(words: List(String)) -> Maybe(String)
  List.head(words)
end
06

Conditionals

if returns a value, like Ruby's, so it always needs the else, because there is no implicit nil branch. case is for shapes, if is for booleans.

conditionals.jd 
module Conditionals exposing (abs, sign)

def abs(n: Int) -> Int
  if n < 0 then 0 - n else n end
end


def sign(n: Int) -> Int
  if n < 0 then
    -1
  else
    if n > 0 then 1 else 0 end
  end
end
07

Pipes and lambdas

|> feeds a value into the next call, the way you would chain .map.join in Ruby. A lambda is (x) -> { … }. Leave a _ where an argument goes and you get back a function still waiting for it. add(_, 1) is Ruby's method(:add).curry[1], without the ceremony.

pipes.jd 
module Pipes exposing (add, increments, loud_names)

def loud_names(names: List(String)) -> String
  names
    |> List.map(String.to_upper)
    |> String.join(", ")
end


def add(a: Int, b: Int) -> Int
  a + b
end


def increments(xs: List(Int)) -> List(Int)
  List.map(xs, add(_, 1))
end
08

Maybe and Result

There is no nil. A value that might be missing says so in its type, and you cannot read it without handling the empty case, so NoMethodError on nil has nowhere to come from. Result is the same trick for something that can fail, carrying the reason.

maybes.jd 
module Maybes exposing (halve, label, safe_divide)

def safe_divide(a: Int, b: Int) -> Maybe(Int)
  if b == 0 then Nothing else Just(a / b) end
end


def halve(n: Int) -> Maybe(Int)
  safe_divide(n, 2)
end


def label(n: Int) -> String
  safe_divide(100, n)
    |> Maybe.map(String.from_int)
    |> Maybe.with_default("undefined")
end
09

Interfaces

Closest thing to a Ruby module you include, except you never wire it up. Declare what a type can do, implement it, and every call site resolves to the right implementation while it compiles. ==, compare and ++ work this way already.

interfaces.jd 
module Interfaces exposing (describe, larger)

struct Money = {
  cents: Int
}


interface Show(a) with
  show : a -> String
end


implements Show(Money) with
  show: (m) -> { String.from_int(m.cents) ++ "c" }
end


def describe(m: Money) -> String
  show(m)
end


def larger(a: a, b: a) -> a
  case compare(a, b)
  in GT then a
  else b
  end
end
10

Modules and imports

One module per file, and the path has to match the name. Shop.Cart lives in shop/cart.jd, the way Zeitwerk expects. exposing is the public surface; everything else stays private.

imports.jd 
module Imports exposing (lookup, tally)

import Dict exposing (Dict)


def tally(words: List(String)) -> Dict(String, Int)
  List.fold(words, Dict.empty, (acc, w) -> {
    Dict.insert(acc, w, 1)
  })
end


def lookup(counts: Dict(String, Int), word: String) -> Int
  counts
    |> Dict.get(word)
    |> Maybe.with_default(0)
end
11

Reaching Ruby

Nothing in Jade touches the outside world on its own. A uses block declares what you want from Ruby, you write that bit in Ruby, and the call hands back a Task that has not run yet, so a signature tells you whether a function does IO. Nothing implements this port here; the console will say so, and you can write one into it.

ports.jd 
module Ports exposing (welcome)

uses Mailer with
  deliver : String -> Task(Bool, String)
end


def welcome(email: String) -> Task(Bool, String)
  deliver(email)
end
