Values and primitives

Int, Float, String, Char and Bool. Annotations go on the signature; inside a function the compiler works the rest out. A zero-argument function is a value where you reference it — total, not total(), though the () stays at the definition so it is greppable.

module Values exposing (greeting, initial, pi, ready, total)

def total() -> Int
  6 * 7
end


def pi() -> Float
  3.14159
end


def greeting() -> String
  "hello" ++ ", " ++ "world"
end


def initial() -> Char
  'J'
end


def ready() -> Bool
  total > 0
end

Functions

Every function declares its argument and return types. ++ joins strings and lists alike, through the Appendable interface.

module Functions exposing (full_name, shout)

def full_name(first: String, last: String) -> String
  first ++ " " ++ last
end


def shout(name: String) -> String
  name
    |> full_name("hey")
    |> String.to_upper
end

Union types

A type is one of a fixed set of shapes, each carrying whatever it needs. Adding a variant is how you find every place that has to care.

module Unions exposing (describe, radius_of)

type Shape
  = Circle(Float)
  | Rectangle(Float, Float)
  | Point


def describe(shape: Shape) -> String
  case shape
  in Circle(_) then "circle"
  in Rectangle(_, _) then "rectangle"
  in Point then "point"
  end
end


def radius_of(shape: Shape) -> Maybe(Float)
  case shape
  in Circle(r) then Just(r)
  in Rectangle(_, _) then Nothing
  in Point then Nothing
  end
end

Structs and records

Named records with typed fields. { p | age: … } produces a new value with one field replaced — the result keeps its type.

module Structs exposing (birthday, label, rename)

struct Person = {
  name: String,
  age: Int
}


def birthday(p: Person) -> Person
  { p | age: p.age + 1 }
end


def rename(p: Person, name: String) -> Person
  { p | name: name }
end


def label(p: Person) -> String
  p.name ++ " (" ++ String.from_int(p.age) ++ ")"
end

Pattern matching

case destructures as it branches, and the compiler rejects a match that forgets a case. Keyed variant payloads read as records; lists split head from rest.

module Matching exposing (first_word, sum, total_of)

type Payment
  = Cash(Int)
  | Card(last4: String, cents: Int)


def total_of(p: Payment) -> Int
  case p
  in Cash(c) then c
  in Card(d) then d.cents
  end
end


def sum(xs: List(Int)) -> Int
  case xs
  in [] then 0
  in [x | rest] then x + sum(rest)
  end
end


def first_word(words: List(String)) -> Maybe(String)
  List.head(words)
end

Conditionals

if is an expression, so it always needs its else. Use case for data shapes and if for booleans.

module Conditionals exposing (abs, sign)

def abs(n: Int) -> Int
  if n < 0 then 0 - n else n end
end


def sign(n: Int) -> Int
  if n < 0 then
    -1
  else
    if n > 0 then 1 else 0 end
  end
end

Pipes, lambdas, placeholders

|> feeds a value into the next call. A lambda is (x) -> { … }, and _ in an argument position curries the call — add(_, 1) is a function of one argument.

module Pipes exposing (add, increments, loud_names)

def loud_names(names: List(String)) -> String
  names
    |> List.map(String.to_upper)
    |> String.join(", ")
end


def add(a: Int, b: Int) -> Int
  a + b
end


def increments(xs: List(Int)) -> List(Int)
  List.map(xs, add(_, 1))
end

Maybe and Result

Absence is a type rather than a value, so there is no nil to forget. Result is the same idea carrying a reason for the failure.

module Maybes exposing (halve, label, safe_divide)

def safe_divide(a: Int, b: Int) -> Maybe(Int)
  if b == 0 then Nothing else Just(a / b) end
end


def halve(n: Int) -> Maybe(Int)
  safe_divide(n, 2)
end


def label(n: Int) -> String
  safe_divide(100, n)
    |> Maybe.map(String.from_int)
    |> Maybe.with_default("undefined")
end

Interfaces

Like typeclasses: the compiler picks the implementation from the argument's type, at compile time. Eq, Comparable and Appendable ship built in, and you can declare your own.

module Interfaces exposing (describe, larger)

struct Money = {
  cents: Int
}


interface Show(a) with
  show : a -> String
end


implements Show(Money) with
  show: (m) -> { String.from_int(m.cents) ++ "c" }
end


def describe(m: Money) -> String
  show(m)
end


def larger(a: a, b: a) -> a
  case compare(a, b)
  in GT then a
  else b
  end
end

Modules and imports

One module per file, and the file name has to imply the module name — Shop.Cart lives in shop/cart.jd. exposing lists the public surface.

module Imports exposing (lookup, tally)

import Dict exposing (Dict)


def tally(words: List(String)) -> Dict(String, Int)
  List.fold(words, Dict.empty, (acc, w) -> {
    Dict.insert(acc, w, 1)
  })
end


def lookup(counts: Dict(String, Int), word: String) -> Int
  counts
    |> Dict.get(word)
    |> Maybe.with_default(0)
end

Reaching Ruby

Jade has no implicit side effects. A uses block declares a port, its return type is a Task, and nothing runs until something runs it.

module Interop exposing (welcome)

uses Mailer with
  deliver : String -> Task(Bool, String)
end


def welcome(email: String) -> Task(Bool, String)
  deliver(email)
end
